#+vet explicit-allocators
package domain

Status :: enum { Ok, Waiting }
Maybe_Value :: union { int }
limit :: 3
foreign_text :: "foreign import is only text"

present :: proc() -> bool { return true }
lookup :: proc() -> (int, bool) { return 1, true }
status :: proc() -> Status { return .Ok }
optional :: proc() -> Maybe_Value { return nil }

@(private)
internal :: proc() -> Domain_Failure { return .None }

acknowledge :: proc() {
	local: State
	_ = advance(&local)
	_ = store()
}
